enable_testing()
get_filename_component(plot_framework_root "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
get_target_property(plot_framework_binary eui_neo BINARY_DIR)

file(GLOB plot_unit_sources CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/unit/*.cpp")
foreach(source IN LISTS plot_unit_sources)
    get_filename_component(name "${source}" NAME_WE)
    add_executable(${name} "${source}")
    set_target_properties(${name} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${plot_framework_binary}")
    eui_apply_compile_options(${name})
    target_link_libraries(${name} PRIVATE eui::module_plot)
    target_compile_definitions(${name} PRIVATE EUI_TEST_SOURCE_DIR="${plot_framework_root}")
    add_test(NAME ${name} COMMAND ${name})
    set_tests_properties(${name} PROPERTIES LABELS "unit")
endforeach()

set_tests_properties(plot_benchmark plot_spatial_benchmark PROPERTIES LABELS "benchmark" RUN_SERIAL TRUE)
foreach(count IN ITEMS 100000 1000000)
    add_test(NAME plot_benchmark_${count} COMMAND plot_benchmark ${count})
    set_tests_properties(plot_benchmark_${count} PROPERTIES LABELS "benchmark" RUN_SERIAL TRUE)
endforeach()
foreach(step IN ITEMS 0.16 0.04)
    add_test(NAME plot_spatial_benchmark_${step} COMMAND plot_spatial_benchmark ${step})
    set_tests_properties(plot_spatial_benchmark_${step} PROPERTIES LABELS "benchmark" RUN_SERIAL TRUE)
endforeach()
foreach(scene IN ITEMS slice isosurface)
    add_test(NAME plot_spatial_benchmark_${scene} COMMAND plot_spatial_benchmark 0.08 ${scene})
    set_tests_properties(plot_spatial_benchmark_${scene} PROPERTIES LABELS "benchmark" RUN_SERIAL TRUE)
endforeach()

file(GLOB plot_probe_sources CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/probes/*.cpp")
foreach(source IN LISTS plot_probe_sources)
    get_filename_component(name "${source}" NAME_WE)
    if(EUI_RESOLVED_RENDER_BACKEND STREQUAL "vulkan" AND
       NOT name STREQUAL "plot_vulkan_probe" AND NOT name STREQUAL "plot_interaction_benchmark")
        continue()
    endif()
    # Vulkan probe 的源码要求 Vulkan；OpenGL 配置只注册对应后端的窗口测试。
    if(EUI_RESOLVED_RENDER_BACKEND STREQUAL "opengl" AND name STREQUAL "plot_vulkan_probe")
        continue()
    endif()
    add_executable(${name} "${source}")
    set_target_properties(${name} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${plot_framework_binary}")
    eui_apply_compile_options(${name})
    target_link_libraries(${name} PRIVATE eui::module_plot)
    target_compile_definitions(${name} PRIVATE EUI_TEST_SOURCE_DIR="${plot_framework_root}")
    eui_neo_copy_assets(${name})
    eui_neo_copy_runtime_dependencies(${name})
    if(MSVC)
        target_link_options(${name} PRIVATE /SUBSYSTEM:CONSOLE)
    elseif(MINGW)
        target_link_options(${name} PRIVATE -mconsole)
    endif()
    if(name STREQUAL "plot_interaction_benchmark")
        add_test(NAME ${name} COMMAND ${name})
    else()
        add_test(NAME ${name} COMMAND ${name} --frames 3)
    endif()
    set_tests_properties(${name} PROPERTIES LABELS "probe" RUN_SERIAL TRUE)
    if(name MATCHES "^plot_gallery_")
        set_tests_properties(${name} PROPERTIES LABELS "probe;benchmark")
    endif()
    if(name STREQUAL "plot_vulkan_probe")
        set_tests_properties(${name} PROPERTIES FAIL_REGULAR_EXPRESSION "Validation Error;VUID-;SYNC-HAZARD")
    endif()
    if(name STREQUAL "plot_interaction_benchmark")
        add_test(NAME plot_interaction_benchmark_125k COMMAND plot_interaction_benchmark 125000)
        set_tests_properties(plot_interaction_benchmark_125k PROPERTIES LABELS "benchmark" RUN_SERIAL TRUE)
    endif()
endforeach()
