cmake_minimum_required(VERSION 3.14)
project(PlotBinarySdkExample LANGUAGES CXX)
enable_testing()
find_package(EuiNeo CONFIG REQUIRED)
if(NOT TARGET eui::module_plot)
    message(FATAL_ERROR "This SDK does not contain eui::module_plot")
endif()
add_executable(plot_data_example data.cpp)
target_compile_features(plot_data_example PRIVATE cxx_std_17)
target_link_libraries(plot_data_example PRIVATE eui::module_plot)
add_test(NAME plot_data_example COMMAND plot_data_example)
add_executable(plot_view_example app.cpp)
target_compile_features(plot_view_example PRIVATE cxx_std_17)
eui_neo_configure_app(plot_view_example)
target_link_libraries(plot_view_example PRIVATE eui::module_plot)
